import matplotlib.pyplot as plt
from google.colab import files  # for automatic download in Colab

# Data
categories = ['Medium Risk', 'High Risk', 'Low Risk']
percentages = [64.3, 7.1, 28.6]

# Colours: Medium = Orange, High = Red, Low = Green
colours = ['#fdae61', '#d73027', '#91cf60']

# Create figure
fig, ax = plt.subplots(figsize=(8, 8))

# Pie chart with custom colours and larger text
wedges, texts, autotexts = ax.pie(
    percentages,
    labels=categories,
    autopct='%1.1f%%',
    colors=colours,
    startangle=180,
    textprops={'fontsize': 16, 'weight': 'bold'},
    wedgeprops={'edgecolor': 'white', 'linewidth': 2}
)

# Make percentage text larger and bolder
for autotext in autotexts:
    autotext.set_fontsize(18)
    autotext.set_weight('bold')
    autotext.set_color('white')

# Make category labels larger
for text in texts:
    text.set_fontsize(16)
    text.set_weight('bold')

# Ensure pie is a circle
ax.axis('equal')

# No title

# Save and download
plt.tight_layout()
plt.savefig('Risk_Category.png', dpi=600, bbox_inches='tight')
plt.show()
files.download('Risk_Category.png')
