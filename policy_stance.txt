import matplotlib.pyplot as plt
from google.colab import files  # for automatic download in Colab

# Data
stances = ['Supportive', 'Conditional', 'Cautious/Restrictive']
counts = [6, 6, 2]

# Colours: Blue gradient
colours = ['#6baed6', '#2171b5', '#08306b']

# Create figure
fig, ax = plt.subplots(figsize=(10, 6))

# Vertical bar chart
bars = ax.bar(stances, counts, color=colours, width=0.6, edgecolor='black', linewidth=1.5)

# Add count labels above each bar
ax.bar_label(bars, padding=5, fontsize=16, weight='bold')

# Set y-axis limits
ax.set_ylim(0, 8)

# Set y-axis label (x-axis label remains empty)
ax.set_ylabel('Number of Institutions', fontsize=16, weight='bold')
ax.set_xlabel('')

# Customise tick fonts
ax.tick_params(axis='both', labelsize=14)

# No title

# Add light grid for readability
ax.grid(axis='y', linestyle=':', alpha=0.5)

# Save and download
plt.tight_layout()
plt.savefig('Policy_Stance.png', dpi=600, bbox_inches='tight')
plt.show()
files.download('Policy_Stance.png')
